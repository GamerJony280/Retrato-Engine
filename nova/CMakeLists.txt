function(nova_demo target source expected)
 nova_add_module(${target}_module "${source}")
 add_executable(${target} examples/native_main.cpp)
 add_dependencies(${target} ${target}_module)
 target_compile_definitions(${target} PRIVATE NOVA_MODULE_HEADER="${${target}_module_SOURCE}" NOVA_EXPECT_X=${expected})
 target_link_libraries(${target} PRIVATE nova_support)
endfunction()
