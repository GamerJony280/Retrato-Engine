cmake_minimum_required(VERSION 3.20)
project(RetratoGame LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
option(GAME_USE_NOVA "Move the cube with Nova instead of C++" OFF)
option(GAME_USE_LUA "Add a Lua actor in the same world" OFF)
if(NOT TARGET retrato::app)
 if(RETRATO_SOURCE_DIR)
  if(NOT EXISTS "${RETRATO_SOURCE_DIR}/include/ember/application.hpp")
   message(FATAL_ERROR "This template needs Retrato 0.7+; the selected source tree is older")
  endif()
  set(EMBER_APP ON CACHE BOOL "Portable application")
  set(EMBER_TOOLS OFF CACHE BOOL "Developer tools")
  set(EMBER_EXAMPLES OFF CACHE BOOL "Engine examples")
  set(EMBER_TESTS OFF CACHE BOOL "Engine tests")
  set(EMBER_IMPORTERS OFF CACHE BOOL "Use cooked assets")
  set(EMBER_VIEWER OFF CACHE BOOL "Legacy desktop viewers")
  set(EMBER_LUA ${GAME_USE_LUA} CACHE BOOL "Lua support")
  set(EMBER_NOVA ${GAME_USE_NOVA} CACHE BOOL "Nova compiler")
  if(GAME_USE_LUA)
   set(EMBER_LUA ON CACHE BOOL "Lua requested by game" FORCE)
  endif()
  if(GAME_USE_NOVA)
   set(EMBER_NOVA ON CACHE BOOL "Nova requested by game" FORCE)
  endif()
  add_subdirectory("${RETRATO_SOURCE_DIR}" engine)
 else()
  find_package(Retrato 0.7 CONFIG REQUIRED)
 endif()
endif()
retrato_add_game(retrato_game SOURCES main.cpp CONTENT_DIR "${CMAKE_CURRENT_SOURCE_DIR}/content")
if(GAME_USE_NOVA)
 if(NOT TARGET nova_support AND NOT TARGET retrato::nova_support)
  message(FATAL_ERROR "Nova is optional: rebuild the SDK with EMBER_NOVA=ON")
 endif()
 nova_add_module(starter_script movement.nova)
 add_dependencies(retrato_game starter_script)
 target_compile_definitions(retrato_game PRIVATE GAME_NOVA_HEADER="${starter_script_SOURCE}")
 target_link_libraries(retrato_game PRIVATE retrato::nova_support)
endif()
if(GAME_USE_LUA)
 if(TARGET ember)
  get_target_property(_game_has_lua ember RETRATO_LUA_ENABLED)
 else()
  set(_game_has_lua ${Retrato_WITH_LUA})
 endif()
 if(NOT _game_has_lua)
  message(FATAL_ERROR "Lua is optional: rebuild the SDK with EMBER_LUA=ON")
 endif()
 target_compile_definitions(retrato_game PRIVATE GAME_USE_LUA)
endif()
